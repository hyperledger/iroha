seiyaku Conditions {
    fn check(int x, bool found) -> int {
        if (x > 1) {
            return 1;
        } else if ((x < 0)) {
            return 2;
        }
        if ! found { return 3; }
        if !(x < 4) { return 4; }
        if (x > 1) && found { return 5; }
        let int y = if (found) { 1 } else { 2 };
        return - y;
    }
}
